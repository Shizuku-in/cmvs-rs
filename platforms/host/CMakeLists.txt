cmake_minimum_required(VERSION 3.19)
project(cmvs_host_native LANGUAGES C CXX)

if(NOT (APPLE OR ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "Linux"))
    message(FATAL_ERROR "CMVS host supports macOS, iOS, Android and Linux")
endif()
get_filename_component(CMVS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_program(CMVS_CARGO cargo REQUIRED)
find_program(CMVS_RUSTC rustc REQUIRED)

set(cmvs_env "RUSTC=${CMVS_RUSTC}" "CARGO_PROFILE_RELEASE_DEBUG=1")
if(ANDROID)
    set(cmvs_abi "${CMAKE_ANDROID_ARCH_ABI}")
    if(NOT cmvs_abi)
        set(cmvs_abi "${ANDROID_ABI}")
    endif()
    if(cmvs_abi STREQUAL "arm64-v8a")
        set(cmvs_target aarch64-linux-android)
        set(cmvs_clang_target aarch64-linux-android)
    elseif(cmvs_abi STREQUAL "armeabi-v7a")
        set(cmvs_target armv7-linux-androideabi)
        set(cmvs_clang_target armv7a-linux-androideabi)
    elseif(cmvs_abi STREQUAL "x86_64")
        set(cmvs_target x86_64-linux-android)
        set(cmvs_clang_target x86_64-linux-android)
    elseif(cmvs_abi STREQUAL "x86")
        set(cmvs_target i686-linux-android)
        set(cmvs_clang_target i686-linux-android)
    else()
        message(FATAL_ERROR "Unsupported CMVS Android ABI: ${cmvs_abi}")
    endif()
    set(cmvs_android_api "${ANDROID_PLATFORM_LEVEL}")
    if(NOT cmvs_android_api)
        set(cmvs_android_api "${CMAKE_SYSTEM_VERSION}")
    endif()
    if(cmvs_android_api LESS 26)
        message(FATAL_ERROR "CMVS native audio requires Android API 26 or later (AAudio)")
    endif()
    get_filename_component(cmvs_ndk_bin "${CMAKE_C_COMPILER}" DIRECTORY)
    set(cmvs_cc "${cmvs_ndk_bin}/${cmvs_clang_target}${cmvs_android_api}-clang")
    set(cmvs_cxx "${cmvs_ndk_bin}/${cmvs_clang_target}${cmvs_android_api}-clang++")
    if(CMAKE_HOST_WIN32)
        string(APPEND cmvs_cc ".cmd")
        string(APPEND cmvs_cxx ".cmd")
    endif()
    if(NOT EXISTS "${cmvs_cc}" OR NOT EXISTS "${cmvs_cxx}")
        message(FATAL_ERROR "CMVS requires the Android NDK compiler wrappers: ${cmvs_cc}")
    endif()
    string(TOUPPER "${cmvs_target}" cmvs_cargo_target)
    string(REPLACE "-" "_" cmvs_cargo_target "${cmvs_cargo_target}")
    list(APPEND cmvs_env
        "CC_${cmvs_target}=${cmvs_cc}" "CXX_${cmvs_target}=${cmvs_cxx}"
        "AR_${cmvs_target}=${CMAKE_AR}"
        "CARGO_TARGET_${cmvs_cargo_target}_LINKER=${cmvs_cc}")
else()
    set(cmvs_arch "${CMAKE_SYSTEM_PROCESSOR}")
    if(APPLE AND CMAKE_OSX_ARCHITECTURES)
        set(cmvs_arch "${CMAKE_OSX_ARCHITECTURES}")
    endif()
    if(cmvs_arch STREQUAL "arm64" OR cmvs_arch STREQUAL "aarch64")
        set(cmvs_cpu aarch64)
    elseif(cmvs_arch STREQUAL "x86_64" OR cmvs_arch STREQUAL "AMD64")
        set(cmvs_cpu x86_64)
    else()
        message(FATAL_ERROR "Build CMVS once per supported architecture: ${cmvs_arch}")
    endif()
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        string(TOLOWER "${CMAKE_OSX_SYSROOT}" cmvs_sdk)
        if(cmvs_sdk MATCHES "iphonesimulator" AND cmvs_cpu STREQUAL "aarch64")
            set(cmvs_target aarch64-apple-ios-sim)
        else()
            set(cmvs_target "${cmvs_cpu}-apple-ios")
        endif()
    elseif(APPLE)
        set(cmvs_target "${cmvs_cpu}-apple-darwin")
    else()
        set(cmvs_target "${cmvs_cpu}-unknown-linux-gnu")
    endif()
endif()

if(NOT DEFINED CMVS_CARGO_TARGET_DIR)
    set(CMVS_CARGO_TARGET_DIR "${CMAKE_CURRENT_BINARY_DIR}/cargo-target" CACHE PATH "CMVS Cargo build cache")
endif()
set(cmvs_library "${CMVS_CARGO_TARGET_DIR}/${cmvs_target}/release/libcmvs_host.a")
file(GLOB_RECURSE cmvs_inputs CONFIGURE_DEPENDS
    "${CMVS_ROOT}/crates/*/src/*.rs" "${CMVS_ROOT}/crates/*/Cargo.toml"
    "${CMVS_ROOT}/crates/*/build.rs"
    "${CMVS_ROOT}/crates/cmvs-core/profiles/*.txt"
    "${CMVS_ROOT}/crates/cmvs-core/profiles/*.profile"
    "${CMVS_ROOT}/tools/*.py"
    "${CMVS_ROOT}/docs/games/chronoclock/contract-*.json")
list(APPEND cmvs_inputs "${CMVS_ROOT}/Cargo.toml" "${CMVS_ROOT}/Cargo.lock")
# The compositor needs optimization even in a Debug application. Keep symbols.
add_custom_command(OUTPUT "${cmvs_library}"
    COMMAND "${CMAKE_COMMAND}" -E env ${cmvs_env}
        "CARGO_TARGET_DIR=${CMVS_CARGO_TARGET_DIR}"
        "${CMVS_CARGO}" build --locked --manifest-path "${CMVS_ROOT}/Cargo.toml"
        -p cmvs-host --lib --release --target "${cmvs_target}"
    DEPENDS ${cmvs_inputs} WORKING_DIRECTORY "${CMVS_ROOT}" VERBATIM USES_TERMINAL)
add_custom_target(cmvs_host_build ALL DEPENDS "${cmvs_library}")
add_library(cmvs_host STATIC IMPORTED GLOBAL)
set_target_properties(cmvs_host PROPERTIES
    IMPORTED_LOCATION "${cmvs_library}"
    INTERFACE_INCLUDE_DIRECTORIES "${CMVS_ROOT}/crates/cmvs-host")
add_dependencies(cmvs_host cmvs_host_build)
if(APPLE)
    foreach(framework AVFAudio AudioToolbox CoreAudio CoreFoundation Foundation Security SystemConfiguration)
        find_library(CMVS_${framework}_FRAMEWORK ${framework} REQUIRED)
        target_link_libraries(cmvs_host INTERFACE "${CMVS_${framework}_FRAMEWORK}")
    endforeach()
elseif(ANDROID)
    target_link_libraries(cmvs_host INTERFACE aaudio android log ${CMAKE_DL_LIBS} m)
else()
    find_package(ALSA REQUIRED)
    find_package(Threads REQUIRED)
    target_link_libraries(cmvs_host INTERFACE ALSA::ALSA Threads::Threads ${CMAKE_DL_LIBS} m)
endif()
message(STATUS "CMVS host Rust target: ${cmvs_target}")
